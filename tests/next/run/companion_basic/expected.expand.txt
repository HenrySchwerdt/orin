import { helper } from "./annos";

// generated for `@Json` on `User`, `Account`

/**
 * A pair of ints.
 * Generated for every annotated record.
 */
export type Pair = { left: int; right: int; };

export type Shape = Circle(int) | Dot;

/**
 * The greeting.
 */
export const GREETING: string = "hi";

export function describeUser(u: User): string {
  return "user " + u.id.toString() + ": " + u.name;
}

function __g10000_sum(accounts: Account[]): int {
  let sum = 0;
  for (const a of accounts) {
    sum = sum + a.balance;
  }
  return sum;
}

/**
 * Sums the balances.
 */
export function totalBalance(accounts: Account[]): int {
  return __g10000_sum(accounts) + helper(5) + secretBump(1);
}

export function shapeArea(s: Shape): int {
  return match (s) {
    Circle(r) => r * r * 3,
    Dot => 0,
  };
}

export class Counter {
  private n: int;
  constructor(start: int) {
    this.n = start;
  }
  bump(by: int): int {
    this.n = this.n + by;
    return this.n;
  }
}

export function countdown(n: int): int {
  let i = n;
  let steps: int = 0;
  while (i > 0) {
    i = i - 1;
    steps = steps + 1;
  }
  return steps;
}

export function classify(n: int): string {
  if (n < 0) {
    return "negative";
  } else {
    if (n == 0) {
      return "zero";
    }
  }
  return !(n > 100) ? "positive" : "huge";
}

export function secretCode(): int {
  const s = Secret { code: 7 };
  return s.code;
}

export function firstOr(xs: int[], d: int): int {
  if (xs.length == 0) {
    return d;
  }
  return xs[0];
}

export function orZero(o: Option<int>): int {
  return match (o) {
    Some(v) => v,
    None => 0,
  };
}

export function doubled(r: Result<int, string>): Result<int, string> {
  const v = r?;
  return Ok(v * 2);
}

export function firstTwo(n: int): int[] {
  return [n, -n, -3];
}

export function kind(n: int): string {
  match (n) {
    1 => {
      return "one";
    },
    _ => {
      return "many";
    },
  }
}
