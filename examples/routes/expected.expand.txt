// companion module ./handlers.gen

import { splitPath } from "./route";

// generated for `@Route` on `Users`

export function routeTable(): string[] {
  return [
    "GET /users/:id -> Users.find",
    "POST /users -> Users.create",
    "GET /users/:id/posts/:post -> Users.post",
  ];
}

export function matchRoute(verb: string, path: string): string {
  const parts = splitPath(path);
  if (verb == "GET" && parts.length == 2 && parts[0] == "users") {
    return "Users.find" + (" id=" + parts[1]);
  }
  if (verb == "POST" && parts.length == 1 && parts[0] == "users") {
    return "Users.create";
  }
  if (verb == "GET" && parts.length == 4 && parts[0] == "users" && parts[2] == "posts") {
    return "Users.post" + (" id=" + parts[1]) + (" post=" + parts[3]);
  }
  return "";
}
