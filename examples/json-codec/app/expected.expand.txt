// companion module ./models.gen

import {
  JsonValue, jNull, jInt, jBool, jStr, jArr, jField, jObj, isNull, decInt, decBool, decString,
  decArray, decObject, lookup, required, indexPath,
} from "json-codec";

// generated for `@Json` on `Address`, `User`

export function encodeAddress(v: Address): JsonValue {
  return jObj([jField("street", jStr(v.street)), jField("zip", jInt(v.zip))]);
}

export function decodeAddressAt(j: JsonValue, path: string): Result<Address, string> {
  const o = decObject(j, path)?;
  const f_street = decString(required(o, "street", path)?, path + ".street")?;
  const f_zip = decInt(required(o, "zip", path)?, path + ".zip")?;
  return Ok(Address { street: f_street, zip: f_zip });
}

export function decodeAddress(j: JsonValue): Result<Address, string> {
  return decodeAddressAt(j, "$");
}

export function encodeUser(v: User): JsonValue {
  return jObj([
    jField("id", jInt(v.id)),
    jField("display_name", jStr(v.name)),
    jField("admin", jBool(v.admin)),
    jField("tags", __g10000_encArrOfstring(v.tags)),
    jField("nick", __g10000_encOptOfstring(v.nick)),
    jField("address", encodeAddress(v.address)),
    jField("previous", __g10000_encArrOfAddress(v.previous)),
    jField("scores", __g10000_encOptOfArrOfint(v.scores)),
  ]);
}

export function decodeUserAt(j: JsonValue, path: string): Result<User, string> {
  const o = decObject(j, path)?;
  const f_id = decInt(required(o, "id", path)?, path + ".id")?;
  const f_name = decString(required(o, "display_name", path)?, path + ".display_name")?;
  const f_admin = decBool(required(o, "admin", path)?, path + ".admin")?;
  const f_tags = __g10000_decArrOfstring(required(o, "tags", path)?, path + ".tags")?;
  let f_nick: Option<string> = None;
  const r_nick = lookup(o, "nick");
  if (r_nick.isSome()) {
    f_nick = __g10000_decOptOfstring(r_nick.unwrap(), path + ".nick")?;
  }
  const f_address = decodeAddressAt(required(o, "address", path)?, path + ".address")?;
  const f_previous = __g10000_decArrOfAddress(required(o, "previous", path)?, path + ".previous")?;
  let f_scores: Option<int[]> = None;
  const r_scores = lookup(o, "scores");
  if (r_scores.isSome()) {
    f_scores = __g10000_decOptOfArrOfint(r_scores.unwrap(), path + ".scores")?;
  }
  return Ok(User {
    id: f_id,
    name: f_name,
    admin: f_admin,
    tags: f_tags,
    nick: f_nick,
    address: f_address,
    previous: f_previous,
    scores: f_scores,
  });
}

export function decodeUser(j: JsonValue): Result<User, string> {
  return decodeUserAt(j, "$");
}

function __g10000_encArrOfstring(v: string[]): JsonValue {
  let out: JsonValue[] = [];
  for (const x of v) {
    out.push(jStr(x));
  }
  return jArr(out);
}

function __g10000_decArrOfstring(j: JsonValue, path: string): Result<string[], string> {
  const items = decArray(j, path)?;
  let out: string[] = [];
  let i: usize = 0;
  while (i < items.length) {
    const e = decString(items[i], indexPath(path, i))?;
    out.push(e);
    i = i + 1;
  }
  return Ok(out);
}

function __g10000_encOptOfstring(v: Option<string>): JsonValue {
  return match (v) {
    Some(x) => jStr(x),
    None => jNull(),
  };
}

function __g10000_decOptOfstring(j: JsonValue, path: string): Result<Option<string>, string> {
  if (isNull(j)) {
    return Ok(None);
  }
  const x = decString(j, path)?;
  return Ok(Some(x));
}

function __g10000_encArrOfAddress(v: Address[]): JsonValue {
  let out: JsonValue[] = [];
  for (const x of v) {
    out.push(encodeAddress(x));
  }
  return jArr(out);
}

function __g10000_decArrOfAddress(j: JsonValue, path: string): Result<Address[], string> {
  const items = decArray(j, path)?;
  let out: Address[] = [];
  let i: usize = 0;
  while (i < items.length) {
    const e = decodeAddressAt(items[i], indexPath(path, i))?;
    out.push(e);
    i = i + 1;
  }
  return Ok(out);
}

function __g10000_encArrOfint(v: int[]): JsonValue {
  let out: JsonValue[] = [];
  for (const x of v) {
    out.push(jInt(x));
  }
  return jArr(out);
}

function __g10000_decArrOfint(j: JsonValue, path: string): Result<int[], string> {
  const items = decArray(j, path)?;
  let out: int[] = [];
  let i: usize = 0;
  while (i < items.length) {
    const e = decInt(items[i], indexPath(path, i))?;
    out.push(e);
    i = i + 1;
  }
  return Ok(out);
}

function __g10000_encOptOfArrOfint(v: Option<int[]>): JsonValue {
  return match (v) {
    Some(x) => __g10000_encArrOfint(x),
    None => jNull(),
  };
}

function __g10000_decOptOfArrOfint(j: JsonValue, path: string): Result<Option<int[]>, string> {
  if (isNull(j)) {
    return Ok(None);
  }
  const x = __g10000_decArrOfint(j, path)?;
  return Ok(Some(x));
}
